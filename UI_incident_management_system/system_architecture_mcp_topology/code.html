Prisma Schema & MCP Entity Binding

schema.prisma 
datasource db {
  provider = "sqlite"
  url      = "file:./data/incident.db"
}

model Incident {
  id          String   @id @default(cuid())
  ticketNo    String   @unique
  title       String
  severity    Severity @default(MEDIUM)
  status      Status   @default(TRIAGE)
  createdAt   DateTime @default(now())
  
  // MCP Embedding & Retrieval Hook
  cleanText   String?  // PII-sanitized buffer
  attachments Attachment[]
}

model Attachment {
  id          String   @id @default(cuid())
  incidentId  String
  fileName    String
  mimeType    String
  incident    Incident @relation(fields: [incidentId], references: [id])
}
verified The cleanText column is populated during Next.js server action ingestion. When MCP daemon initiates semantic search, it executes direct vectorized reads without mutating the primary SQLite table.
